# pixi-silk: examples

> Smooth vector graphics for PixiJS v8. Every shape is an exact signed distance field, anti-aliased per pixel in the shader, so edges stay smooth at any zoom, rotation and device pixel ratio without MSAA.

pixi-silk 0.1.0 is an MIT-licensed TypeScript library. It needs pixi.js ^8.6 as a peer dependency, and a browser with WebGL2 or WebGPU.

Site: https://pixi-silk.schmooky.dev
Repository: https://github.com/schmooky/pixi-silk
Package: https://www.npmjs.com/package/pixi-silk

This file holds whole screens built with pixi-silk, with the full source of every demo. The demos run in the docs demo runner: `defineDemo({ setup({ stage, params, tick }) })`. In your app, `stage` is `app.stage` (or a container on it), `params` are your own values and `tick(fn)` is `app.ticker.add`.

- [llms-full.txt](https://pixi-silk.schmooky.dev/llms-full.txt): core, paint and rendering: guides and API, SilkGraphics first.
- [llms-ui.txt](https://pixi-silk.schmooky.dev/llms-ui.txt): text, layout, UI widgets and the editor kit: guides and API.

### Weather, in two tones

Demo sources run in the docs demo runner, `defineDemo({ setup({ stage, params, tick }) })`. In your own app, `stage` is `app.stage` (or a container on it), `params` holds the values of the demo controls, and `tick(fn)` is `app.ticker.add`, with `t` and `dt` in seconds.

Live demo `weather-mono` (source):

```ts
import { Container, Text, type TextStyleFontWeight } from 'pixi.js';
import { damp, SilkGraphics } from 'pixi-silk';
import { defineDemo } from './runtime';

// A monochrome weather dashboard. Static shapes redraw only while the theme fades, the needle,
// compass and sunrise arrow move on their own layer every frame.

const W = 390;
const H = 900;
const MONO = '"JetBrains Mono", "SF Mono", Menlo, Consolas, monospace';

interface Palette {
    bg: number;
    card: number;
    disc: number;
    onDisc: number;
    ink: number;
    faint: number;
}

const LIGHT: Palette = {
    bg: 0xdcdedb,
    card: 0xebecea,
    disc: 0x1d1d1d,
    onDisc: 0xf2f2f2,
    ink: 0x1d1d1d,
    faint: 0x9a9c99,
};
const DARK: Palette = {
    bg: 0x000000,
    card: 0x1e1f21,
    disc: 0xe9eaea,
    onDisc: 0x111111,
    ink: 0xf2f2f2,
    faint: 0x6a6b6d,
};

const mix = (a: number, b: number, t: number) => {
    const ch = (s: number) => Math.round(((a >> s) & 255) + (((b >> s) & 255) - ((a >> s) & 255)) * t) << s;

    return ch(16) | ch(8) | ch(0);
};
const blend = (t: number): Palette => ({
    bg: mix(LIGHT.bg, DARK.bg, t),
    card: mix(LIGHT.card, DARK.card, t),
    disc: mix(LIGHT.disc, DARK.disc, t),
    onDisc: mix(LIGHT.onDisc, DARK.onDisc, t),
    ink: mix(LIGHT.ink, DARK.ink, t),
    faint: mix(LIGHT.faint, DARK.faint, t),
});

// layout: two columns of cards under the hero row
const COL = [20, 205];
const CARD_W = 165;
const CARD_H = 165;
const ROW = [330, 515, 700];
const hair = (color: number, alpha = 1) => ({ width: 1.25, color, alpha });

type Role = 'ink' | 'onDisc' | 'faint';

export default defineDemo({
    size: [W, H],
    controls: {
        theme: { type: 'select', label: 'theme', options: ['light', 'dark'], value: 'dark' },
    },
    setup({ stage, params, tick }) {
        const base = new SilkGraphics();
        const live = new SilkGraphics();
        const texts = new Container();
        const labels: [Text, Role][] = [];
        let fade = params.theme === 'dark' ? 1 : 0;
        let drawnFade = -1;

        stage.addChild(base, live, texts);

        const text = (
            value: string,
            x: number,
            y: number,
            size: number,
            role: Role,
            anchor = 0,
            weight: TextStyleFontWeight = '400',
        ) => {
            const t = new Text({
                text: value,
                style: {
                    fill: 0xffffff,
                    fontFamily: MONO,
                    fontSize: size,
                    fontWeight: weight,
                    letterSpacing: size * 0.08,
                },
            });

            t.anchor.set(anchor, 0.5);
            t.position.set(x, y);
            texts.addChild(t);
            labels.push([t, role]);
        };

        // header
        text('BROADGATE', W / 2, 92, 17, 'ink', 0.5);
        // hero
        text('12°', 115, 204, 66, 'onDisc', 0.5, '300');
        text('H:14° L:9°', 115, 254, 13, 'onDisc', 0.5);
        text('AIR QUALITY  21', 228, 162, 13, 'ink');
        text('Fair', 228, 184, 13, 'ink');
        text('UV INDEX 02', 228, 254, 13, 'ink');
        text('Low', 228, 276, 13, 'ink');
        // card titles and values
        const card = (col: number, row: number, title: string, sub: string, value: string, unit = '') => {
            const x = COL[col] + 16;
            const y = ROW[row];

            text(title, x, y + 22, 11, 'ink');
            if (sub) text(sub, x, y + 38, 11, 'ink', 0, '700');
            text(value, x, y + CARD_H - 26, 26, 'ink', 0, '300');
            if (unit) text(unit, x + value.length * 17 + 4, y + CARD_H - 22, 10, 'ink');
        };

        card(0, 0, 'AIR QUALITY', 'Fair', '21');
        card(1, 0, 'HUMIDITY', '', '79%');
        card(0, 1, 'WIND SPEED', '', '7', 'MPH');
        card(1, 1, 'UV INDEX', 'Low', '02');
        card(0, 2, 'SUNRISE', '', '06:11');
        text('N', COL[0] + CARD_W / 2, ROW[1] + 56, 9, 'ink', 0.5, '700');
        text('FEELS LIKE', COL[1] + CARD_W / 2, ROW[2] + 60, 12, 'onDisc', 0.5);
        text('11°', COL[1] + CARD_W / 2, ROW[2] + 100, 46, 'onDisc', 0.5, '300');

        const drawBase = (p: Palette) => {
            const g = base.clear();

            g.rect(0, 0, W, H).fill(p.bg);
            menuIcon(g, 28, 92, p.ink);
            gearIcon(g, W - 38, 92, p);
            g.circle(115, 218, 90).fill(p.disc);
            g.pill(212, 130, 160, 74).fill(p.card);
            g.pill(212, 222, 160, 74).fill(p.card);
            g.line(20, 318, W - 20, 318).stroke(hair(p.ink));
            for (const [col, row] of [
                [0, 0],
                [1, 0],
                [0, 1],
                [1, 1],
                [0, 2],
            ]) {
                g.roundRect(COL[col], ROW[row], CARD_W, CARD_H, 14).fill(p.card);
            }
            gaugeFace(g, COL[0] + CARD_W / 2, ROW[0] + 108, p);
            g.circle(COL[1] + CARD_W / 2, ROW[0] + 82, 46).stroke(hair(p.ink, 0.6));
            g.circle(COL[1] + CARD_W / 2, ROW[0] + 82, 37).fill(p.disc);
            compassFace(g, COL[0] + CARD_W / 2, ROW[1] + 92, p);
            uvStack(g, COL[1] + CARD_W / 2, ROW[1] + 120, p);
            sunriseFace(g, COL[0] + CARD_W / 2, ROW[2] + 92, p);
            g.circle(COL[1] + CARD_W / 2, ROW[2] + CARD_H / 2, 82).fill(p.disc);
        };

        tick((t, dt) => {
            fade = damp(fade, params.theme === 'dark' ? 1 : 0, 6, dt);
            const p = blend(fade);

            if (Math.abs(fade - drawnFade) > 0.001) {
                drawBase(p);
                drawnFade = fade;
                for (const [label, role] of labels) label.tint = p[role];
            }
            const g = live.clear();

            gaugeNeedle(g, COL[0] + CARD_W / 2, ROW[0] + 108, 0.28 + Math.sin(t * 0.7) * 0.04, p.ink);
            compassNeedle(g, COL[0] + CARD_W / 2, ROW[1] + 92, 0.45 + Math.sin(t * 0.9) * 0.08, p.onDisc);
            sunriseArrow(g, COL[0] + CARD_W / 2, ROW[2] + 92, Math.sin(t * 1.6) * 2.5, p.ink);
        });

        return undefined;
    },
});

function menuIcon(g: SilkGraphics, x: number, y: number, color: number) {
    for (let i = -1; i <= 1; i++) {
        g.circle(x, y + i * 7, 1.5).fill(color);
        g.line(x + 6, y + i * 7, x + 24, y + i * 7).stroke({ width: 1.5, color, cap: 'round' });
    }
}

function gearIcon(g: SilkGraphics, x: number, y: number, p: Palette) {
    g.star(x, y, 8, 13, 10.5, 0, 1.2).fill(p.ink);
    g.circle(x, y, 8.5).fill(p.bg);
    g.circle(x, y, 4).fill(p.ink);
    g.circle(x, y, 2.6).fill(p.bg);
}

function gaugeFace(g: SilkGraphics, cx: number, cy: number, p: Palette) {
    const r = 50;

    g.line(cx - 72, cy, cx + 72, cy).stroke(hair(p.ink));
    g.arc(cx, cy, r, Math.PI, Math.PI * 2).stroke(hair(p.ink));
    for (let i = 0; i <= 16; i++) {
        const a = Math.PI + (i / 16) * Math.PI;
        const inner = i % 4 === 0 ? r - 9 : r - 5;

        g.line(cx + Math.cos(a) * inner, cy + Math.sin(a) * inner, cx + Math.cos(a) * r, cy + Math.sin(a) * r).stroke(
            hair(p.ink),
        );
    }
}

function gaugeNeedle(g: SilkGraphics, cx: number, cy: number, value: number, color: number) {
    const a = Math.PI + value * Math.PI;
    const tipX = cx + Math.cos(a) * 44;
    const tipY = cy + Math.sin(a) * 44;
    const head = 7;

    g.line(cx, cy, tipX, tipY).stroke({ width: 1.5, color, cap: 'round' });
    for (const side of [-0.45, 0.45]) {
        g.line(tipX, tipY, tipX - Math.cos(a + side) * head, tipY - Math.sin(a + side) * head).stroke({
            width: 1.5,
            color,
            cap: 'round',
        });
    }
}

function compassFace(g: SilkGraphics, cx: number, cy: number, p: Palette) {
    g.circle(cx, cy, 44).stroke(hair(p.ink));
    g.circle(cx, cy, 30).fill(p.disc);
    g.regularPoly(cx, cy - 47, 3.5, 3, 0, 0.5).fill(p.ink);
}

function compassNeedle(g: SilkGraphics, cx: number, cy: number, angle: number, color: number) {
    const dx = Math.sin(angle) * 24;
    const dy = -Math.cos(angle) * 24;

    g.line(cx + dx, cy + dy, cx - dx, cy - dy).stroke({ width: 1.4, color, cap: 'round' });
    const back = Math.atan2(-dy, -dx);

    for (const side of [-0.5, 0.5]) {
        g.line(cx - dx, cy - dy, cx - dx - Math.cos(back + side) * 6, cy - dy - Math.sin(back + side) * 6).stroke({
            width: 1.4,
            color,
            cap: 'round',
        });
    }
}

function uvStack(g: SilkGraphics, cx: number, cy: number, p: Palette) {
    // `cy` is the shared base line: each ellipse sits on it, the tallest faintest
    for (let i = 0; i < 5; i++) {
        const ry = 36 - i * 6.5;

        g.ellipse(cx, cy - ry, 46, ry).stroke(hair(p.ink, 0.22 + i * 0.13));
    }
    g.ellipse(cx, cy, 46, 4.5).stroke({ width: 1.8, color: p.ink });
}

function sunriseFace(g: SilkGraphics, cx: number, cy: number, p: Palette) {
    g.line(cx - 72, cy, cx + 72, cy).stroke(hair(p.ink));
    g.arc(cx, cy, 52, Math.PI, Math.PI * 2).stroke(hair(p.ink, 0.45));
    g.arc(cx, cy, 40, Math.PI, Math.PI * 2).stroke({ width: 1.6, color: p.ink });
    g.line(cx - 52, cy + 10, cx + 52, cy + 10).stroke(hair(p.ink, 0.7));
    g.line(cx - 34, cy + 18, cx + 34, cy + 18).stroke(hair(p.ink, 0.7));
    g.line(cx - 12, cy + 26, cx + 12, cy + 26).stroke(hair(p.ink, 0.7));
}

function sunriseArrow(g: SilkGraphics, cx: number, cy: number, lift: number, color: number) {
    const top = cy - 26 - lift;

    g.line(cx, cy - 8, cx, top).stroke({ width: 1.4, color, cap: 'round' });
    g.line(cx, top, cx - 5, top + 6).stroke({ width: 1.4, color, cap: 'round' });
    g.line(cx, top, cx + 5, top + 6).stroke({ width: 1.4, color, cap: 'round' });
}
```

Hairlines, discs and one monospace face. The needle, compass and sunrise arrow drift on their own layer.

```ts
// A finance bento board. Every tile is its own container: it slides in on load, lifts on hover,

const COL_W = (W - GAP * 4) / 3;

const SANS = 'Inter, "Helvetica Neue", system-ui, sans-serif';

const C = {
    page: 0x2a2a2a,
    orange: 0xf28a2e,
    orangeDeep: 0xd9762a,
    ink: 0x111111,
    stone: 0x8f8f8b,
    mist: 0xd9d9d6,
    paper: 0xf2f2f0,
    olive: 0x7f8b6c,
    oliveDeep: 0x5e6950,
    slate: 0x5a5a5a,
    arrow: 0xc4c4c4,
    clay: 0x7d5546,
    taupe: 0x857a5c,
    sand: 0xc8b88d,
    donut: 0x4a4a4a,
};

interface Tile {
    view: Container;
    /** 0..1 entrance progress, starts after `delay`. */
    enter: number;
    delay: number;
    lift: number;
    hover: boolean;
    /** Redraws the animated parts with the eased entrance value. */
    update?: (p: number, t: number) => void;
}

const label = (
    text: string,
    size: number,
    color: number,
    weight: TextStyleFontWeight = '500',
    extra: TextStyleOptions = {},
) => new Text({ text, style: { fill: color, fontFamily: SANS, fontSize: size, fontWeight: weight, ...extra } });

function tile(x: number, y: number, w: number, h: number, fill: number, delay: number, radius = 10): Tile {
    const view = new Container();
    const bg = new SilkGraphics();

    bg.roundRect(0, 0, w, h, radius).fill(fill);
    view.addChild(bg);
    view.position.set(x, y);
    view.eventMode = 'static';
    view.hitArea = { contains: (px, py) => px >= 0 && py >= 0 && px <= w && py <= h };

    const t: Tile = { view, enter: 0, delay, lift: 0, hover: false };

    view.on('pointerover', () => (t.hover = true));
    view.on('pointerout', () => (t.hover = false));

    return t;
}

function place(child: Container, x: number, y: number, ax = 0, ay = 0) {
    if ('anchor' in child) (child as Text).anchor.set(ax, ay);
    child.position.set(x, y);

    return child;
}
```

```ts
/** Blue to red to white, the palette of the hot spot. */
function hot(v: number): number {
    const stops = [0x1a2cff, 0xff1e1e, 0xffffff];
    const x = Math.min(1, Math.max(0, v)) * (stops.length - 1);
    const i = Math.min(stops.length - 2, Math.floor(x));
    const t = x - i;
    const ch = (s: number) => Math.round(((stops[i] >> s) & 255) * (1 - t) + ((stops[i + 1] >> s) & 255) * t) << s;

    return ch(16) | ch(8) | ch(0);
}

function drawCells(g: SilkGraphics, t: number) {
    g.clear();
    const hotX = 0.22 + Math.sin(t * 0.4) * 0.05;
    const hotY = -0.18 + Math.cos(t * 0.33) * 0.04;

    for (let j = 0; j < GRID; j++) {
        for (let i = 0; i < GRID; i++) {
            const x = CX - RADIUS + i * CELL;
            const y = CY - RADIUS + j * CELL;
            const u = (x + CELL / 2 - CX) / RADIUS;
            const v = (y + CELL / 2 - CY) / RADIUS;

            if (u * u + v * v > 1) continue;
            // a soft grey body, brighter towards its lower right, with a slow ripple
            const body =
                Math.max(0, 1 - Math.hypot((u - 0.05) / 0.9, (v - 0.15) / 0.75)) *
                (0.6 + 0.4 * Math.sin(u * 6 + t * 0.7));

                Math.max(0, 1 - Math.hypot((u - hotX) / 0.42, (v - hotY) / 0.28)) * 1.25 +
                Math.max(0, 0.6 - Math.hypot(u - 0.2, v + 0.55) * 1.6);

            if (heat > 0.18) g.rect(x, y, CELL - 1, CELL - 1).fill(hot(heat - 0.18));
            else if (body > 0.04) g.rect(x, y, CELL - 1, CELL - 1).fill({ color: grey(body), alpha: 0.85 });
        }
```

# Slot HUD examples

URL: https://pixi-silk.schmooky.dev/docs/slot-hud/
Area: Examples

> Copyable slot HUD controls built from pixi-silk atoms: a spin button with autoplay, win meters, multiplier badges, free-spin pips, feature paths and hold-to-buy.

These controls are examples, not part of the library. Each demo is one file: open its source,
copy it into your game and restyle it. They are built from the same atoms as the library widgets:
`SilkGraphics`, `Spring` and `damp`, signals, `trackGesture` and `RadialMenu`.

### Spin button and autoplay

Demo sources run in the docs demo runner, `defineDemo({ setup({ stage, params, tick }) })`. In your own app, `stage` is `app.stage` (or a container on it), `params` holds the values of the demo controls, and `tick(fn)` is `app.ticker.add`, with `t` and `dt` in seconds.

Live demo `slot-spin` (source):

```ts
import { Container, Text } from 'pixi.js';
import { damp, SilkGraphics, Spring } from 'pixi-silk';
import { signal } from 'pixi-silk/signal';
import { type GesturePoint, trackGesture } from 'pixi-silk/ui/gesture';
import { type RadialItem, RadialMenu } from 'pixi-silk/ui/radial';
import { defineDemo } from './runtime';

// A slot spin button built from the atoms (copy it into your game and restyle it):
// idle breathing, press squash, hold for the autoplay picker, a stop square while spinning and the
// autoplay count left as a shrinking ring. The picker is a half RadialMenu fanning up from the button.

type Phase = 'idle' | 'spinning' | 'stopping';

class SpinButton extends Container {
    readonly phase = signal<Phase>('idle');
    readonly autoplay = signal(0);
    /** Spins in the current autoplay run, for the ring. */
    autoplayTotal = 1;
    onSpin: () => void = () => {};
    onStop: () => void = () => {};
    onHold: (p: GesturePoint) => void = () => {};

    private readonly g = new SilkGraphics();
    private readonly press = new Spring(0, 700, 30);
    private readonly morph = new Spring(0, 260, 22);
    private spinAngle = 0;
    private ring = 0;
    private t = 0;

    constructor(private readonly radius = 44) {
        super();
        this.addChild(this.g);
        this.hitArea = { contains: (x, y) => Math.hypot(x, y) <= radius + 10 };
        trackGesture(
            this,
            {
                onDown: () => (this.press.target = 1),
                onUp: () => (this.press.target = 0),
                onTap: () => (this.phase.value === 'idle' ? this.onSpin() : this.onStop()),
                onHold: (p) => this.phase.value === 'idle' && this.onHold(p),
            },
            { holdMs: 380, flickSpeed: 0 },
        );
    }

    update(dt: number): void {
        const spinning = this.phase.value !== 'idle';
        const r = this.radius;
        const total = Math.max(1, this.autoplayTotal);

        this.t += dt;
        this.press.step(dt);
        this.morph.target = spinning ? 1 : 0;
        this.morph.step(dt);
        this.spinAngle += dt * (2 + 8 * this.morph.value);
        this.ring = damp(this.ring, this.autoplay.value > 0 ? this.autoplay.value / total : 0, 10, dt);

        const breathe = spinning ? 0 : Math.sin(this.t * 2.4) * 0.025;

        this.scale.set((1 + breathe) * (1 - 0.08 * this.press.value));
        const m = Math.min(1, Math.max(0, this.morph.value));
        const g = this.g.clear();

        g.circle(0, 4, r).fill({ color: 0x000000, alpha: 0.35, blur: 10 });
        g.circle(0, 0, r).fill(spinning ? 0xff453a : 0x30d158);
        g.circle(0, 0, r - 5).stroke({ width: 2, color: 0xffffff, alpha: 0.25 });
        // the arrow turns into a stop square as the reels start
        g.save().rotateTransform(this.spinAngle * (1 - m) * 0.15);
        if (m < 0.5) {
            g.arcSweep(0, 0, r * 0.42, -Math.PI * 0.3, Math.PI * 1.45).stroke({
                width: 7,
                cap: 'round',
                color: 0xffffff,
                alpha: 1 - m * 2,
            });
            g.regularPoly(r * 0.12, -r * 0.42, 9, 3, Math.PI / 2, 2).fill({ color: 0xffffff, alpha: 1 - m * 2 });
        } else {
            const s = r * 0.42 * (m - 0.5) * 2;

            g.roundRect(-s / 2, -s / 2, s, s, 5).fill(0xffffff);
        }
        g.restore();
        if (this.ring > 0.002) {
            g.arcSweep(0, 0, r + 8, -Math.PI / 2, Math.PI * 2 * this.ring).stroke({
                width: 4,
                cap: 'round',
                color: 0xffd60a,
            });
        }
    }
}

const chip = (n: string): RadialItem => ({ id: n, label: n });

export default defineDemo({
    size: [520, 320],
    touchAction: 'none',
    setup({ silk, stage, tick }) {
        const spin = new SpinButton();
        const status = new Text({
            text: 'tap to spin, hold for autoplay',
            style: { fill: 0x8e8e93, fontFamily: 'system-ui, sans-serif', fontSize: 13 },
        });
        const picker = new RadialMenu({
            items: ['10', '25', '50', '100', '∞'].map(chip),
            sweep: Math.PI * 0.95,
            direction: -Math.PI / 2,
            innerRadius: 58,
            hub: false,
            outerRadius: 108,
            ticker: silk.app.ticker,
            theme: { segment: 0x2c2c2e, segmentHover: 0x3a3a3c, icon: 0xf2f2f7, iconHover: 0xffd60a, accent: 0xffd60a },
            onSelect: (item) => startAutoplay(item.id === '∞' ? 999 : Number(item.id)),
        });
        let timer: ReturnType<typeof setTimeout> | null = null;

        status.anchor.set(0.5);
        status.position.set(260, 30);
        spin.position.set(260, 250);
        picker.position.set(260, 250);
        stage.addChild(status, spin, picker);

        const spinOnce = () => {
            spin.phase.set('spinning');
            status.text = spin.autoplay.value > 0 ? `autoplay: ${spin.autoplay.value} left` : 'spinning';
            timer = setTimeout(() => {
                spin.phase.set('idle');
                if (spin.autoplay.value > 0) {
                    spin.autoplay.update((n) => n - 1);
                    if (spin.autoplay.value > 0) timer = setTimeout(spinOnce, 250);
                    else status.text = 'autoplay done';
                } else status.text = 'tap to spin, hold for autoplay';
            }, 1100);
        };
        const startAutoplay = (n: number) => {
            spin.autoplayTotal = n;
            spin.autoplay.set(n);
            spinOnce();
        };

        spin.onSpin = spinOnce;
        spin.onStop = () => {
            spin.autoplay.set(0);
            status.text = 'stopped';
        };
        spin.onHold = (p) => picker.openAt(260, 250, p);
        tick((_t, dt) => spin.update(dt));

        return () => {
            if (timer) clearTimeout(timer);
        };
    },
});
```

The arrow morphs into a stop square while the reels run. The yellow ring shows the autoplay spins
left. Holding opens a half `RadialMenu` in marking mode, so press, hold, drag and release picks a
count in one gesture.

```ts
            ticker,
            radius: 62,
            value: betIndex.value,
            min: 0,
            max: LADDER.length - 1,
            step: 1,
            format: (i) => `$${LADDER[Math.round(i)]}`,
            onCommit: (i) => betIndex.set(Math.round(i)),
        });
        const spin = new SilkGraphics();
        const spinText = new Text({
            text: 'SPIN',
            style: { fill: 0xffffff, fontFamily: 'system-ui, sans-serif', fontWeight: '700', fontSize: 16 },
        });

        panel.roundRect(30, 30, 260, 120, 20, 0.6).fill(0x1c1c1e);
        panel.roundRect(310, 30, 260, 120, 20, 0.6).fill(0x1c1c1e);
        balance.position.set(160, 100);
        win.position.set(440, 100);
        bet.position.set(160, 250);
        spin.position.set(440, 250);
        spinText.anchor.set(0.5);
        spinText.position.set(440, 250);
        stage.addChild(
            panel,
            label('BALANCE', 160, 55),
            label('WIN', 440, 55),
            label('BET', 160, 330),
            balance,
            win,
            bet,
            spin,
            spinText,
        );

        const drawSpin = () => {
            spin.clear()
                .circle(0, 0, 48)
                .fill(busy.value ? 0x3a3a3c : 0x30d158);
        };

        busy.subscribe(drawSpin);
        drawSpin();
        spin.hitArea = { contains: (x, y) => Math.hypot(x, y) <= 56 };
        trackGesture(spin, {
            onTap: () => {
                const stake = LADDER[betIndex.value];

                if (busy.value || balance.state.value.value < stake) return;
                busy.set(true);
                balance.set(balance.state.value.value - stake);
                setTimeout(() => {
                    const roll = Math.random();
                    const amount = roll < 0.45 ? 0 : roll < 0.9 ? stake * (1 + Math.random() * 4) : stake * 60;

                    if (amount === 0) {
                        busy.set(false);

                        return;
                    }
                    win.set(0);
                    win.skip();
                    win.set(Math.round(amount * 100) / 100);
                }, 600);
            },
        });
        // the win rolls into the balance when it finishes counting (or is skipped)
        win.state.counting.subscribe((counting) => {
            if (counting || !busy.value) return;
            const amount = win.state.value.value;

            setTimeout(() => {
                balance.set(balance.state.value.value + amount);
                win.set(0);
                busy.set(false);
            }, 500);
        });
        win.eventMode = 'static';
        win.hitArea = { contains: (x, y) => Math.abs(x) < 130 && Math.abs(y) < 50 };
        win.on('pointertap', () => win.skip());

        return undefined;
    },
});
```

Built on the library's `Odometer` and `JogDial`. The dial's detents are positions on a bet
ladder, and its readout shows the bet at that position.

### Meters, badges and pips

Live demo `slot-meters` (source):

```ts
import { Text } from 'pixi.js';
import { conic, damp, SilkGraphics, Spring } from 'pixi-silk';
import { signal } from 'pixi-silk/signal';
import { trackGesture } from 'pixi-silk/ui/gesture';
import { defineDemo } from './runtime';

// Slot HUD pieces drawn with one SilkGraphics each, state in signals. Copy and restyle.
// Win meter with tiers, multiplier badge, free-spin pips, feature path and a hold-to-buy button.
// Press "WIN" to feed the meter, hold "BUY" to confirm.

const TIERS = [
    { at: 0.25, name: 'BIG', color: 0x30d158 },
    { at: 0.55, name: 'MEGA', color: 0x0a84ff },
    { at: 0.85, name: 'EPIC', color: 0xbf5af2 },
];
const font = { fontFamily: 'system-ui, sans-serif', fontWeight: '700' } as const;

export default defineDemo({
    size: [640, 360],
    touchAction: 'none',
    setup({ stage, tick }) {
        const win = signal(0); // 0..1 of the max win
        const mult = signal(1);
        const spins = signal(10);
        const used = signal(0);
        const path = signal(0); // feature collect progress, 0..6

        const g = new SilkGraphics();
        const tierText = new Text({ text: '', style: { ...font, fill: 0xffffff, fontSize: 18 } });
        const multText = new Text({ text: 'x1', style: { ...font, fill: 0x1c1c1e, fontSize: 22 } });
        const winBtn = new Text({ text: 'WIN', style: { ...font, fill: 0xffffff, fontSize: 14 } });
        const buyText = new Text({ text: 'HOLD TO BUY', style: { ...font, fill: 0xffffff, fontSize: 13 } });

        // one ramp over the whole gauge, so the colour says how far the win got
        const meterPaint = conic([0x30d158, 0x0a84ff, 0xbf5af2, 0xff375f], {
            center: [150, 170],
            units: 'local',
            startAngle: Math.PI * 0.75,
            sweep: Math.PI * 1.5,
        });
        let shown = 0;
        let tierPop = 0;
        let lastTier = -1;
        const badge = new Spring(1, 520, 14);
        let shake = 0;
        let pipPop = new Array(10).fill(0);
        let pathShown = 0;
        let buy = 0;
        let holding = false;
        let bought = 0;
        let time = 0;

        for (const t of [tierText, multText, winBtn, buyText]) t.anchor.set(0.5);
        tierText.position.set(150, 190);
        multText.position.set(380, 112);
        winBtn.position.set(150, 300);
        buyText.position.set(520, 300);
        stage.addChild(g, tierText, multText, winBtn, buyText);

        mult.subscribe(() => {
            badge.value = 1.45;
            shake = 1;
        });
        used.subscribe((u) => (pipPop[u - 1] = 1));

        // hit zones: a container per control keeps the example readable
        const zone = (x: number, y: number, w: number, h: number) => {
            const z = new SilkGraphics();

            z.hitArea = { contains: (px, py) => px >= x && px <= x + w && py >= y && py <= y + h };
            stage.addChild(z);

            return z;
        };

        trackGesture(zone(90, 280, 120, 40), {
            onTap: () => {
                win.update((v) => (v >= 1 ? 0 : Math.min(1, v + 0.12 + Math.random() * 0.1)));
                if (Math.random() < 0.5) mult.update((m) => (m >= 10 ? 1 : m + 1));
                if (used.value < spins.value) used.update((u) => u + 1);
                else {
                    used.set(0);
                    pipPop = pipPop.map(() => 0);
                }
                path.update((p) => (p >= 6 ? 0 : p + 1));
            },
        });
        trackGesture(
            zone(440, 280, 160, 40),
            {
                onDown: () => (holding = true),
                onUp: () => (holding = false),
            },
            { holdMs: 0, flickSpeed: 0 },
        );

        tick((_t, dt) => {
            time += dt;
            shown = damp(shown, win.value, 4, dt);
            badge.target = 1;
            badge.step(dt);
            shake = damp(shake, 0, 6, dt);
            pipPop = pipPop.map((p) => damp(p, 0, 5, dt));
            pathShown = damp(pathShown, path.value, 6, dt);
            buy = holding ? Math.min(1, buy + dt / 1.2) : damp(buy, 0, 10, dt);
            if (buy >= 1 && holding) {
                holding = false;
                bought = 1;
                buy = 0;
            }
            bought = damp(bought, 0, 2.5, dt);

            const tier = TIERS.reduce((acc, t, i) => (shown >= t.at ? i : acc), -1);

            if (tier !== lastTier) {
                if (tier > lastTier) tierPop = 1;
                lastTier = tier;
            }
            tierPop = damp(tierPop, 0, 5, dt);

            g.clear();
            // win meter: a 270deg gauge with tier ticks; the fill takes the colour of the tier reached
            const cx = 150;
            const cy = 170;
            const start = Math.PI * 0.75;
            const sweep = Math.PI * 1.5;

            g.arcSweep(cx, cy, 90, start, sweep).stroke({ width: 16, cap: 'round', color: 0x2c2c2e });
            if (shown > 0.003) {
                g.arcSweep(cx, cy, 90, start, sweep * shown).stroke({
                    width: 16,
                    cap: 'round',
                    gradient: meterPaint,
                });
            }
            for (const t of TIERS) {
                const a = start + sweep * t.at;

                g.line(
                    cx + Math.cos(a) * 76,
                    cy + Math.sin(a) * 76,
                    cx + Math.cos(a) * 104,
                    cy + Math.sin(a) * 104,
                ).stroke({
                    width: 2,
                    color: shown >= t.at ? t.color : 0x48484a,
                });
            }
            if (tier >= 0) {
                g.circle(cx, cy + 20, 34 + tierPop * 20).fill({
                    color: TIERS[tier].color,
                    alpha: 0.25 + tierPop * 0.4,
                    blur: 14,
                });
            }
            tierText.text = tier >= 0 ? `${TIERS[tier].name} WIN` : `${Math.round(shown * 100)}%`;
            tierText.scale.set(1 + tierPop * 0.35);

            // multiplier badge: a rounded star that pops and shakes when the value changes
            const bx = 380 + Math.sin(time * 60) * 4 * shake;

            g.star(bx, 112, 8, 46 * badge.value, 38 * badge.value, time * 0.4, 4).fill({
                color: 0xffd60a,
                blur: 0,
            });
            g.circle(bx, 112, 56 * badge.value).fill({ color: 0xffd60a, alpha: 0.18 * shake, blur: 12 });
            multText.text = `x${mult.value}`;
            multText.position.x = bx;
            multText.scale.set(badge.value);

            // free-spin pips: used ones fill, the newest one bursts
            for (let i = 0; i < spins.value; i++) {
                const x = 300 + i * 30;
                const y = 220;
                const on = i < used.value;

                g.circle(x, y, 9 + pipPop[i] * 6).fill({ color: on ? 0xff9f0a : 0x2c2c2e });
                if (pipPop[i] > 0.02) {
                    g.circle(x, y, 12 + (1 - pipPop[i]) * 16).stroke({
                        width: 2,
                        color: 0xff9f0a,
                        alpha: pipPop[i],
                    });
                }
            }

            // feature path: collect points along a wave, the trail grows to the reached point
            const pts: number[] = [];

            for (let i = 0; i <= 60; i++) {
                const t = i / 60;

                pts.push(300 + t * 300, 160 + Math.sin(t * Math.PI * 3) * 14);
            }
            g.polyline(pts, { smooth: 'catmull' }).stroke({ width: 6, cap: 'round', color: 0x2c2c2e });
            const reach = Math.round((pathShown / 6) * 60);

            if (reach > 0) {
                g.polyline(pts.slice(0, (reach + 1) * 2), { smooth: 'catmull' }).stroke({
                    width: 6,
                    cap: 'round',
                    color: 0x64d2ff,
                });
            }
            for (let k = 0; k <= 6; k++) {
                const i = k * 10;
                const on = pathShown >= k - 0.01;

                g.circle(pts[i * 2], pts[i * 2 + 1], on ? 8 : 6).fill(on ? 0x64d2ff : 0x48484a);
            }

            // buttons
            g.pill(90, 280, 120, 40).fill(0x0a84ff);
            g.pill(440, 280, 160, 40).fill(bought > 0.05 ? 0x30d158 : 0x3a3a3c);
            if (buy > 0.002) g.pill(440, 280, Math.max(40, 160 * buy), 40).fill({ color: 0xff9f0a, alpha: 0.9 });
            buyText.text = bought > 0.05 ? 'BOUGHT' : 'HOLD TO BUY';
        });

        return undefined;
    },
});
```

### A dock that steps aside

Live demo `spin-dock` (source):

```ts
import { Text } from 'pixi.js';
import { SilkGraphics } from 'pixi-silk';
import { signal } from 'pixi-silk/signal';
import { SilkButton } from 'pixi-silk/ui/button';
import { ButtonDock } from 'pixi-silk/ui/dock';
import { SegmentedSwitch } from 'pixi-silk/ui/modes';
import { defineDemo } from './runtime';

const play = (g: SilkGraphics, s: number) => g.regularPoly(s * 0.06, 0, s * 0.42, 3, Math.PI / 2, 2).fill(0xffffff);
const minus = (g: SilkGraphics, s: number) =>
    g.line(-s * 0.3, 0, s * 0.3, 0).stroke({ width: s * 0.14, cap: 'round', color: 0xffffff });
const plus = (g: SilkGraphics, s: number) => {
    minus(g, s);
    g.line(0, -s * 0.3, 0, s * 0.3).stroke({ width: s * 0.14, cap: 'round', color: 0xffffff });
};
// a lightning bolt as two rounded triangles (closed polygons cannot be filled yet)
const bolt = (g: SilkGraphics, s: number) => {
    g.triangle(s * 0.12, -s * 0.5, -s * 0.3, s * 0.08, s * 0.08, s * 0.02, 1).fill(0xffffff);
    g.triangle(-s * 0.12, s * 0.5, s * 0.3, -s * 0.08, -s * 0.08, -s * 0.02, 1).fill(0xffffff);
};

// Tap SPIN for one spin, hold it for five. While the reels run, the side buttons leave, the speed
// switch locks and the spin button turns busy: nothing can be pressed twice. The bet lives in a signal.
const SPEED: Record<string, number> = { normal: 1.1, turbo: 0.6, quick: 0.3 };
export default defineDemo({
    size: [560, 340],
    touchAction: 'none',
    controls: {
        enabled: { type: 'toggle', label: 'spin enabled', value: true },
    },
    setup({ silk, stage, params, tick }) {
        const ticker = silk.app.ticker;
        const bet = signal(1);
        const reels = new SilkGraphics();
        const label = new Text({
            text: '',
            style: { fill: 0xf2f2f7, fontFamily: 'system-ui, sans-serif', fontSize: 15, fontWeight: '600' },
        });
        const side = { ticker, theme: { fill: 0x2c2c2e, fillHover: 0x3a3a3c, fillPress: 0x1c1c1e } };
        const less = new SilkButton({
            ...side,
            shape: 'circle',
            height: 48,
            icon: minus,
            onPress: () => bet.update((b) => Math.max(1, b - 1)),
        });
        const more = new SilkButton({
            ...side,
            shape: 'circle',
            height: 48,
            icon: plus,
            onPress: () => bet.update((b) => Math.min(10, b + 1)),
        });
        const speed = new SegmentedSwitch({
            ticker,
            value: 'normal',
            segmentWidth: 78,
            height: 34,
            options: [
                { id: 'normal', label: 'Normal' },
                { id: 'turbo', icon: bolt },
                { id: 'quick', label: 'Quick' },
            ],
        });
        const spin = new SilkButton({
            ticker,
            shape: 'circle',
            height: 76,
            icon: play,
            iconSize: 30,
            holdMs: 450,
            theme: { fill: 0x30d158, fillHover: 0x4ade80, fillPress: 0x248a3d },
            onPress: () => run(1),
            onHold: () => run(5),
        });
        const dock = new ButtonDock({ ticker, gap: 14 }).add(less, spin, more);
        let spinning = 0;
        let duration = 1.1;
        let left = 0;

        bet.subscribe((b) => (label.text = `bet ${b}`));
        label.text = `bet ${bet.value}`;
        label.anchor.set(0.5);
        label.position.set(90, 34);
        speed.position.set(400, 34);
        dock.position.set(280, 282);
        stage.addChild(reels, label, speed, dock);

        async function run(count: number) {
            left = count;
            spin.state.busy.set(true);
            speed.state.enabled.set(false);
            await Promise.all([less.hide(), more.hide()]);
            while (left-- > 0) {
                duration = SPEED[speed.state.value.value] ?? 1.1;
                spinning = duration;
                await new Promise((r) => setTimeout(r, duration * 1000));
            }
            spin.state.busy.set(false);
            speed.state.enabled.set(true);
            await dock.showAll();
        }

        tick((t, dt) => {
            spin.state.enabled.set(params.enabled);
            spinning = Math.max(0, spinning - dt);
            reels.clear();
            for (let c = 0; c < 5; c++) {
                const x = 60 + c * 92;
                const run = spinning > 0 && spinning < duration * (1 - c * 0.05) ? 1 : 0;

                reels.roundRect(x, 62, 80, 168, 14, 0.6).fill(0x1c1c1e);
                for (let r = 0; r < 3; r++) {
                    const y = 70 + r * 54 + (run ? ((t * 900 + c * 40) % 54) - 27 : 0);
                    const hue = [0xff453a, 0xffd60a, 0x0a84ff, 0xbf5af2][(r + c + (run ? Math.floor(t * 20) : 0)) % 4];

                    reels.roundRect(x + 12, y + 6, 56, 40, 10, 0.6).fill({ color: hue, alpha: run ? 0.55 : 0.9 });
                }
            }
        });

        return undefined;
    },
});
```

# FAQ

## Are the slot game HUD controls part of pixi-silk?

No. The spin button, autoplay picker, meters and wallet on the slot HUD page are examples. Each is one demo file built from library pieces such as `RadialMenu`, `Odometer` and `JogDial`. Copy the file into your game and restyle it.
